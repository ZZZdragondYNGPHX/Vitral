cmake_minimum_required(VERSION 3.18)
project(vitral-fcitx5 LANGUAGES C CXX)
include(GNUInstallDirs)
find_package(Fcitx5Core REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(VITRAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
find_library(VITRAL_FFI vitral_ffi PATHS "${VITRAL_ROOT}/target/release" "${VITRAL_ROOT}/target/debug" REQUIRED NO_DEFAULT_PATH)
add_library(vitral-fcitx5 MODULE src/engine.cpp)
target_compile_features(vitral-fcitx5 PRIVATE cxx_std_20)
target_compile_definitions(vitral-fcitx5 PRIVATE QT_NO_KEYWORDS)
target_include_directories(vitral-fcitx5 PRIVATE include "${VITRAL_ROOT}/crates/vitral-ffi/include")
target_link_libraries(vitral-fcitx5 PRIVATE Fcitx5::Core Qt6::Core systemd "${VITRAL_FFI}")
set_target_properties(vitral-fcitx5 PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS vitral-fcitx5 DESTINATION "${FCITX_INSTALL_ADDONDIR}")
install(FILES "${VITRAL_FFI}" DESTINATION "${FCITX_INSTALL_ADDONDIR}")
install(FILES addon/vitral.conf DESTINATION "${FCITX_INSTALL_PKGDATADIR}/addon")
install(FILES inputmethod/vitral.conf DESTINATION "${FCITX_INSTALL_PKGDATADIR}/inputmethod")
install(FILES vitral.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus WaylandClient)
find_package(LayerShellQt REQUIRED)
find_package(KWayland REQUIRED)
find_package(KF6WindowSystem REQUIRED)
add_executable(vitral-panel src/panel.cpp)
qt_generate_wayland_protocol_client_sources(vitral-panel PRIVATE_CODE FILES "${CMAKE_CURRENT_SOURCE_DIR}/protocols/ext-data-control-v1.xml")
target_include_directories(vitral-panel PRIVATE include)
target_compile_features(vitral-panel PRIVATE cxx_std_20)
target_link_libraries(vitral-panel PRIVATE Qt6::Widgets Qt6::DBus Qt6::WaylandClient LayerShellQt::Interface KF6::WindowSystem Plasma::KWaylandClient)
install(TARGETS vitral-panel DESTINATION "${CMAKE_INSTALL_BINDIR}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/themes/Vitral/theme.conf" "${CMAKE_CURRENT_BINARY_DIR}/themes/Vitral-Light/theme.conf"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/theme.py" "${VITRAL_ROOT}/tokens/design-tokens.json" "${CMAKE_CURRENT_BINARY_DIR}/themes"
    DEPENDS theme.py "${VITRAL_ROOT}/tokens/design-tokens.json")
add_custom_target(vitral-themes ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/themes/Vitral/theme.conf" "${CMAKE_CURRENT_BINARY_DIR}/themes/Vitral-Light/theme.conf")
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/themes/" DESTINATION "${FCITX_INSTALL_PKGDATADIR}/themes")
